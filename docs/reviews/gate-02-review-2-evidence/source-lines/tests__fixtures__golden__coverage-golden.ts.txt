  25 function days(from: string, to: string, exclude: ReadonlySet<string>): string[] {
  26   const out: string[] = [];
  27   for (let t = Date.parse(`${from}T00:00:00Z`); t < Date.parse(`${to}T00:00:00Z`); t += 86_400_000) {
  28     const iso = new Date(t).toISOString().slice(0, 10);
  29     if (!exclude.has(iso)) out.push(iso);
  30   }
  31   return out;
  32 }
