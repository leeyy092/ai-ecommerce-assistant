   1 /**
   2  * 导入校验 handler（TASK-007 队列边界；正式校验编排属 TASK-008+）。
   3  *
   4  * G2-H05：执行前重查发起人当前有效 Membership 与当前角色类型权限——
   5  *   禁用/降权后遗留任务以 failed+UPLOAD_PERMISSION_REVOKED 终态拒绝，不再解析。
   6  * G2-H08：状态机修复——uploaded/validating → validating 原子推进（重投安全）；
   7  *   文件缺失等持久错误给出失败终态（不把 validating 当成功完成）；
   8  *   暂时性错误交还队列重试，重试耗尽由调用方/清理器落终态。
   9  * 解析经 TASK-006 统一 Adapter 并组装 typed CanonicalBatch（G2-H04 合同消费点）。
  10  * 不做：业务入库、dataset_version 递增、快照评估（TASK-008/013 边界）。
  11  */
  12 import { createHash } from "node:crypto";
  13 import { getPrismaClient } from "@/database/prisma";
  14 import {
  15   createCanonicalBatch,
  16   getAdapter,
  17   isFileKind,
  18   type CanonicalBatch,
  19   type CoverageDeclarationItem,
  20 } from "@/adapters/contracts";
  21 import { canImport } from "@/services/access";
  22 import { getObjectText, putObject } from "@/storage";
  23 import { Readable } from "node:stream";
  24 
  25 export interface ValidateJobData {
  26   taskId: string;
  27 }
  28 
  29 export interface ValidateHandlerOpts {
  30   /** pg-boss 作业当前重试次数/上限（用于判断重试耗尽，落失败终态） */
  31   retryCount?: number;
  32   retryLimit?: number;
  33 }
  34 
  35 export interface ValidateOutcome {
  36   status: string;
  37   valid: number;
  38   errors: number;
  39 }
  40 
  41 /** 终态集合：重投到已终态任务时直接幂等跳过 */
  42 const TERMINAL_STATUSES = new Set(["preview_ready", "failed", "committing", "committed", "expired"]);
  43 
  44 export async function handleValidateTask(
  45   data: ValidateJobData,
  46   opts: ValidateHandlerOpts = {},
  47 ): Promise<ValidateOutcome> {
  48   const db = getPrismaClient();
  49   const task = await db.importTask.findUnique({ where: { id: data.taskId } });
  50   if (!task) return { status: "missing", valid: 0, errors: 0 };
  51   if (!isFileKind(task.sourceKind)) return { status: task.status, valid: 0, errors: 0 };
  52   if (TERMINAL_STATUSES.has(task.status)) {
  53     // 幂等重投：终态任务不重复处理（如 dispatcher 与路由补投竞争产生的重复作业）
  54     return { status: task.status, valid: task.validCount, errors: task.errorCount };
  55   }
  56   if (!task.rawObjectKey) {
  57     // 防御：建账即带文件（G2-H08 窗口1），空键属于不变量破坏 → 明确失败
  58     await db.importTask.updateMany({
  59       where: { id: task.id, status: { in: ["uploaded", "validating"] } },
  60       data: { status: "failed", errorCode: "SOURCE_FILE_MISSING" },
  61     });
  62     return { status: "failed", valid: 0, errors: 0 };
  63   }
  64 
  65   // G2-H05：执行前重查发起人当前有效身份与类型权限（禁用/降权 → 终态拒绝）
  66   const membership = await db.membership.findFirst({
  67     where: { orgId: task.orgId, userId: task.createdBy },
  68     select: { status: true, role: true },
  69   });
  70   if (!membership || membership.status !== "active" || !canImport(membership.role, task.sourceKind)) {
  71     await db.importTask.updateMany({
  72       where: { id: task.id, status: { in: ["uploaded", "validating"] } },
  73       data: { status: "failed", errorCode: "UPLOAD_PERMISSION_REVOKED" },
  74     });
  75     return { status: "failed", valid: 0, errors: 0 };
  76   }
  77 
  78   const store = await db.store.findUniqueOrThrow({
  79     where: { id: task.storeId },
  80     select: { externalStoreId: true },
  81   });
  82   const dataSource = await db.dataSource.findUniqueOrThrow({
  83     where: { id: task.dataSourceId },
  84     select: { sourceNamespace: true },
  85   });
  86 
  87   // uploaded/validating → validating（原子推进；并发重投安全）
  88   await db.importTask.updateMany({
  89     where: { id: task.id, status: { in: ["uploaded", "validating"] } },
  90     data: { status: "validating" },
  91   });
  92 
  93   try {
  94     const text = await getObjectText(task.rawObjectKey);
  95     // G2-H03：来源更新时间必须来自文件本身，缺失即行级错误（不补造）
  96     const parsed = getAdapter("csv").parse(task.sourceKind, { storeExternalId: store.externalStoreId }, text);
  97     // G2-H04：统一 CanonicalBatch 合同——服务端赋值店铺/namespace/checksum，
  98     // 覆盖声明来自任务存档（用户映射确认），不来自文件内容
  99     const batch: CanonicalBatch = createCanonicalBatch({
 100       sourceKind: task.sourceKind,
 101       sourceNamespace: dataSource.sourceNamespace,
 102       storeId: task.storeId,
 103       adapterKind: "csv",
 104       rawChecksum: createHash("sha256").update(text).digest("hex"),
 105       parse: parsed,
 106       coverageDeclaration: Array.isArray(task.coverageDeclaration)
 107         ? (task.coverageDeclaration as unknown as CoverageDeclarationItem[])
 108         : [],
 109     });
 110 
 111     const valid = batch.records.length;
 112     const errors = batch.row_errors.length;
 113     let errorObjectKey: string | null = null;
 114     if (errors > 0) {
 115       errorObjectKey = `errors/${task.id}/errors.csv`;
 116       const header = "row,column,code,message";
 117       const body = batch.row_errors
 118         .map((e) => [e.row, e.column ?? "", e.code, `"${String(e.message).replace(/"/g, '""')}"`].join(","))
 119         .join("\r\n");
 120       await putObject(errorObjectKey, Readable.from([`${header}\r\n${body}\r\n`]));
 121     }
 122 
 123     const failure = batch.row_errors.find((e) =>
 124       ["STORE_MISMATCH", "MISSING_COLUMN", "EMPTY_FILE", "INVALID_CSV", "DUPLICATE_COLUMN"].includes(e.code),
 125     );
 126     await db.importTask.updateMany({
 127       where: { id: task.id, status: "validating" },
 128       data: {
 129         status: failure ? "failed" : "preview_ready",
 130         validCount: valid,
 131         errorCount: errors,
 132         errorObjectKey,
 133         errorCode: failure ? failure.code : null,
 134         previewVersion: { increment: 1 },
 135       },
 136     });
 137     return { status: failure ? "failed" : "preview_ready", valid, errors };
 138   } catch (error) {
 139     const code = (error as { code?: string }).code;
 140     const exhausted =
 141       opts.retryCount !== undefined && opts.retryLimit !== undefined && opts.retryCount >= opts.retryLimit;
 142     if (code === "ENOENT" || code === "ENOTDIR" || exhausted) {
 143       // G2-H08：持久失败给明确终态（文件缺失/对象损坏/重试耗尽），不留悬挂 validating
 144       await db.importTask.updateMany({
 145         where: { id: task.id, status: "validating" },
 146         data: { status: "failed", errorCode: code === "ENOENT" || code === "ENOTDIR" ? "SOURCE_FILE_MISSING" : "VALIDATE_FAILED" },
 147       });
 148       return { status: "failed", valid: 0, errors: 0 };
 149     }
 150     // 暂时性错误：交还队列重试（pg-boss 按 retryLimit 重投）
 151     throw error;
 152   }
 153 }
